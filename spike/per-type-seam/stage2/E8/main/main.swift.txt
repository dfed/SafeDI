nonisolated(unsafe) var nextIdentifier = 0
nonisolated(unsafe) var built = 0

func identifier() -> Int {
	nextIdentifier += 1
	return nextIdentifier
}

func walk(_ value: Home) {
	built += 1
	walk(value.screen0Builder.instantiate(identifier()))
	walk(value.screen1Builder.instantiate(identifier()))
	walk(value.screen2Builder.instantiate(identifier()))
	walk(value.screen3Builder.instantiate(identifier()))
}

func walk(_ value: Screen0) {
	built += 1
	walk(value.model0)
	walk(value.screen5Builder.instantiate(identifier()))
	walk(value.screen6Builder.instantiate(identifier()))
	walk(value.screen7Builder.instantiate(identifier()))
}

	built += 1

func walkEveryPath(_ root: Root) {
	walk(root.home)
}

let arguments = CommandLine.arguments
if arguments.count > 1, arguments[1] == "bench" {
	// One round: builds a Root and walks every path, again until the walks last at least 100 ms; prints nanoseconds per
	// walk, a walk being a Root plus every path, so that a variant's cost per tree build is in the figure.
	var walks = 0
	let clock = ContinuousClock()
	walkEveryPath(Root())
	let start = clock.now
	var elapsed = Duration.zero
	while elapsed < .milliseconds(100) {
		walkEveryPath(Root())
		walks += 1
		elapsed = clock.now - start
	}
	let (seconds, attoseconds) = elapsed.components
	let nanoseconds = Double(seconds) * 1e9 + Double(attoseconds) / 1e9
	print("\(nanoseconds / Double(walks)) ns per walk, \(walks) walks, \(built / (walks + 1)) values per walk")
} else {
	do {
		let root = Root()
		walkEveryPath(root)
		print("trace of every path:")
		Registry.printTrace()
		withExtendedLifetime(root) {}
	}
	Registry.printAlive("after dropping everything")
}
