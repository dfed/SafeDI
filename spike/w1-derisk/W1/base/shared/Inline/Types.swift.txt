import Harness
import SafeDI

public protocol Api: AnyObject {}

@Instantiable
public final class Screen: Instantiable, Traced {
    public init(api: any Api) {
        self.api = api
        Registry.created(self, ["api": api])
    }

    public let tracker = Tracker("Screen")
    @Instantiated public let api: any Api
}

@Instantiable
public final class Home: Instantiable, Traced {
    public init(screenBuilder: Instantiator<Screen>) {
        self.screenBuilder = screenBuilder
        Registry.created(self, ["screenBuilder": screenBuilder])
    }

    public let tracker = Tracker("Home")
    @Instantiated public let screenBuilder: Instantiator<Screen>
}

@Instantiable(fulfillingAdditionalTypes: [(any Api).self])
public final class LiveApi: Instantiable, Traced, Api {
    public init(screenBuilder: Instantiator<Screen>) {
        self.screenBuilder = screenBuilder
        Registry.created(self, ["screenBuilder": screenBuilder])
    }

    public let tracker = Tracker("LiveApi")
    @Received public let screenBuilder: Instantiator<Screen>
}

@Instantiable(isRoot: true)
public final class Root: Instantiable, Traced {
    public init(home: Home) {
        self.home = home
        Registry.created(self, ["home": home])
    }

    public let tracker = Tracker("Root")
    @Instantiated public let home: Home
}
