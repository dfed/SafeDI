import Harness
import SafeDI

public protocol Api: AnyObject {}

@Instantiable
public final class S3: Instantiable, Traced {
    public init(api: any Api) {
        self.api = api
        Registry.created(self, ["api": api])
    }

    public let tracker = Tracker("S3")
    @Instantiated public let api: any Api
}

@Instantiable
public final class M2: Instantiable, Traced {
    public init() {
        Registry.created(self, [:])
    }

    public let tracker = Tracker("M2")
}

@Instantiable
public final class Badge2: Instantiable, Traced {
    public init(a: S3) {
        self.a = a
        Registry.created(self, ["a": a])
    }

    public let tracker = Tracker("Badge2")
    @Received public let a: S3
}

@Instantiable
public final class S2: Instantiable, Traced {
    public init(m2: M2, a: S3, badge2: Badge2, b: S3) {
        self.m2 = m2
        self.a = a
        self.badge2 = badge2
        self.b = b
        Registry.created(self, ["m2": m2, "a": a, "badge2": badge2, "b": b])
    }

    public let tracker = Tracker("S2")
    @Instantiated public let m2: M2
    @Instantiated public let a: S3
    @Instantiated public let badge2: Badge2
    @Instantiated public let b: S3
}

@Instantiable(fulfillingAdditionalTypes: [(any Api).self])
public final class LiveApi: Instantiable, Traced, Api {
    public init(m1: M1, badge1: Badge1?, m2: M2, badge2: Badge2?) {
        self.m1 = m1
        self.badge1 = badge1
        self.m2 = m2
        self.badge2 = badge2
        Registry.created(self, ["m1": m1, "badge1": badge1, "m2": m2, "badge2": badge2])
    }

    public let tracker = Tracker("LiveApi")
    @Received public let m1: M1
    @Received(onlyIfAvailable: true) public let badge1: Badge1?
    @Received public let m2: M2
    @Received(onlyIfAvailable: true) public let badge2: Badge2?
}
