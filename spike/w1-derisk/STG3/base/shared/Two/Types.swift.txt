import Three
import Harness
import SafeDI

#SafeDIConfiguration(generatePlainDependencyTreeInitializers: true)

@Instantiable
public final class M2: Instantiable, Traced {
    public init() {
        Registry.created(self, [:])
    }

    public let tracker = Tracker("M2")
}

@Instantiable
public final class Badge2: Instantiable, Traced {
    public init(a: S3) {
        self.a = a
        Registry.created(self, ["a": a])
    }

    public let tracker = Tracker("Badge2")
    @Received public let a: S3
}

@Instantiable(generateDependencyTree: true)
public final class S2: Instantiable, Traced {
    public init(m2: M2, a: S3, badge2: Badge2, b: S3) {
        self.m2 = m2
        self.a = a
        self.badge2 = badge2
        self.b = b
        Registry.created(self, ["m2": m2, "a": a, "badge2": badge2, "b": b])
    }

    public let tracker = Tracker("S2")
    @Instantiated public let m2: M2
    @Instantiated public let a: S3
    @Instantiated public let badge2: Badge2
    @Instantiated public let b: S3
}
