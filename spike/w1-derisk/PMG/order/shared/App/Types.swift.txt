    Root.mock()

public func rootMockOverridingMeter() -> Root {
    Root.mock(safeDIOverrides: .init(home: .init(meter: { Meter() })))
}

@Instantiable(fulfillingAdditionalTypes: [(any Api).self])
public final class Live: Instantiable, Traced, Api {
    public init(meter: Meter, badge: Badge?) {
        self.meter = meter
        self.badge = badge
        Registry.created(self, ["meter": meter, "badge": badge])
    }

    public let tracker = Tracker("Live")
    @Received public let meter: Meter
    @Received(onlyIfAvailable: true) public let badge: Badge?
}

@Instantiable(generateMock: true)
public final class Root: Instantiable, Traced {
    public init(home: Home) {
        self.home = home
        Registry.created(self, ["home": home])
    }

    public let tracker = Tracker("Root")
    @Instantiated public let home: Home
}
