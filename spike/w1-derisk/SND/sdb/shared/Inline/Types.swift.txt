import Harness
import SafeDI

public protocol Api: AnyObject, Sendable {}

@Instantiable
public final class A: Instantiable, Traced, Sendable {
    public init(api: any Api) {
        self.api = api
        Registry.created(self, ["api": api])
    }

    public let tracker = Tracker("A")
    @Instantiated public let api: any Api
}

@Instantiable
public final class Db: Instantiable, Traced, Sendable {
    public init() {
        Registry.created(self, [:])
    }

    public let tracker = Tracker("Db")
}

@Instantiable
public final class Store: Instantiable, Traced {
    public init(db: Db, aBuilder: SendableInstantiator<A>) {
        self.db = db
        self.aBuilder = aBuilder
        Registry.created(self, ["db": db, "aBuilder": aBuilder])
    }

    public let tracker = Tracker("Store")
    @Instantiated public let db: Db
    @Instantiated public let aBuilder: SendableInstantiator<A>
}

@Instantiable(fulfillingAdditionalTypes: [(any Api).self])
public final class LiveApi: Instantiable, Traced, Api, Sendable {
    public init(db: Db) {
        self.db = db
        Registry.created(self, ["db": db])
    }

    public let tracker = Tracker("LiveApi")
    @Received public let db: Db
}

@Instantiable(isRoot: true)
public final class Root: Instantiable, Traced {
    public init(store: Store) {
        self.store = store
        Registry.created(self, ["store": store])
    }

    public let tracker = Tracker("Root")
    @Instantiated public let store: Store
}
