import Feature
import Harness
import SafeDI

@Instantiable(fulfillingAdditionalTypes: [(any Gadget).self])
public final class LiveGadget: Instantiable, Traced, Gadget {
    public init() {
        Registry.created(self, [:])
    }

    public let tracker = Tracker("LiveGadget")
}

@Instantiable(isRoot: true, generateMock: .public)
public final class Root: Instantiable, Traced {
    public init(logger: Logger, service: Service) {
        self.logger = logger
        self.service = service
        Registry.created(self, ["logger": logger, "service": service])
    }

    public let tracker = Tracker("Root")
    @Instantiated public let logger: Logger
    @Instantiated public let service: Service
}
