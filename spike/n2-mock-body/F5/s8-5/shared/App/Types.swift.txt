import Harness

@Instantiable(generateMock: true)
public final class Screen0: Instantiable, Traced {
    public init(id: Int, service: Service, context: Context, screen3Builder: Instantiator<Screen3>, screen5Builder: Instantiator<Screen5>, screen6Builder: Instantiator<Screen6>) {
        self.id = id
        self.service = service
        self.context = context
        self.screen3Builder = screen3Builder
        self.screen5Builder = screen5Builder
        self.screen6Builder = screen6Builder
        Registry.created(self, ["id": id, "service": service, "context": context, "screen3Builder": screen3Builder, "screen5Builder": screen5Builder, "screen6Builder": screen6Builder])
    }

    public let tracker = Tracker("Screen0")
    @Forwarded public let id: Int
    @Received public let service: Service
    @Instantiated public let context: Context
    @Instantiated public let screen3Builder: Instantiator<Screen3>
    @Instantiated public let screen5Builder: Instantiator<Screen5>
    @Instantiated public let screen6Builder: Instantiator<Screen6>
}
