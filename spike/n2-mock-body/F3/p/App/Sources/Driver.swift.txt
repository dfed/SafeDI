import Harness
import SafeDI

func make() -> App { App() }

public func runFixture() {
	do {
		let app = make()
		let preview1 = app.editor.previewBuilder.instantiate()
		let editor2 = preview1.editor.instance
		let preview2 = editor2.previewBuilder.instantiate()
		let editor3 = preview2.editor.instance
		print("trace of three levels:")
		Registry.printTrace()
		withExtendedLifetime((app, preview1, editor2, preview2, editor3)) {}
	}
	Registry.printAlive("after dropping everything")
	do {
		let editor3: Editor
		do {
			editor3 = make().editor.previewBuilder.instantiate().editor.instance.previewBuilder.instantiate().editor.instance
		}
		Registry.printTrace()
		Registry.printAlive("holding only the third editor")
		withExtendedLifetime(editor3) {}
	}
	Registry.printAlive("after dropping it")
}
public let overridesRunner: (@Sendable () -> Void)? = nil
